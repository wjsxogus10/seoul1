# 2. 데이터 생성 헬퍼 (파일 없으면 가짜 데이터 만듦)
# --------------------------------------------------------------------------
def get_mock_data():
    """파일 로드 실패 시 사용할 샘플 데이터 생성"""
    districts = ['강남구', '강동구', '강북구', '강서구', '관악구', '광진구', '구로구', '금천구', '노원구', '도봉구', 
                 '동대문구', '동작구', '마포구', '서대문구', '서초구', '성동구', '성북구', '송파구', '양천구', '영등포구', 
                 '용산구', '은평구', '종로구', '중구', '중랑구']
    
    data = {
        '자치구명': districts,
        '총_상주인구_수': np.random.randint(100000, 600000, size=25),
        '면적(km²)': np.random.uniform(10, 40, size=25),
        '버스정류장_수': np.random.randint(200, 600, size=25),
        '지하철역_수': np.random.randint(3, 20, size=25)
    }
    df = pd.DataFrame(data)
    return df

@st.cache_data(ttl=600)
def load_data_safely():
    status_log = []
    
    # [1] 지도 데이터 (GeoJSON) - 인터넷에서 가져옴
    try:
        map_url = "https://raw.githubusercontent.com/southkorea/seoul-maps/master/kostat/2013/json/seoul_municipalities_geo_simple.json"
        gdf = geopandas.read_file(map_url)
        gdf = gdf.to_crs(epsg=4326)
        
        # 컬럼 이름 통일
        if 'name' in gdf.columns: gdf = gdf.rename(columns={'name': '자치구명'})
        elif 'SIG_KOR_NM' in gdf.columns: gdf = gdf.rename(columns={'SIG_KOR_NM': '자치구명'})
        
        status_log.append("✅ 지도 데이터 로드 성공")
    except Exception as e:
        status_log.append(f"❌ 지도 데이터 실패 (기본값 사용): {e}")
        # 지도가 없으면 빈 껍데기라도 만듦
        gdf = pd.DataFrame({'자치구명': ['강남구']*25}) # 임시

    # [2] CSV 데이터 로드 시도
    # 파일명 정의
    file_pop = '서울시 상권분석서비스(상주인구-자치구).csv'
    file_bus = '20250930기준_서울시정류소(정류소별).xlsx - 시내버스 정류장.csv'
    file_sub = '서울교통공사_자치구별 지하철역 정보_20250804.CSV'

    # 파일 찾기 함수
    def find_file(fname):
        if os.path.exists(fname): return fname
        if os.path.exists(f'./data/{fname}'): return f'./data/{fname}'
        return None

    path_pop = find_file(file_pop)
    path_bus = find_file(file_bus)
    path_sub = find_file(file_sub)

    # 데이터프레임 초기화
    df_main = get_mock_data() # 일단 가짜 데이터로 채워둠 (덮어쓰기 위해)

    # --- 인구 병합 ---
    if path_pop:
        try:
            df = pd.read_csv(path_pop, encoding='cp949')
            # 숫자 변환
            if '총_상주인구_수' in df.columns:
                grp = df.groupby('자치구_코드_명')['총_상주인구_수'].mean().reset_index().rename(columns={'자치구_코드_명':'자치구명'})
                # 기존 가짜 데이터 삭제하고 진짜로 교체
                df_main = df_main.drop(columns=['총_상주인구_수'], errors='ignore')
                df_main = df_main.merge(grp, on='자치구명', how='left')
                status_log.append(f"✅ 인구 파일 적용됨 ({path_pop})")
        except: status_log.append("❌ 인구 파일 읽기 오류")
    else:
        status_log.append("⚠️ 인구 파일 없음 (샘플 데이터 사용)")

    # --- 버스 병합 ---
    if path_bus:
        try:
            try: df = pd.read_csv(path_bus, encoding='utf-8')
            except: df = pd.read_csv(path_bus, encoding='cp949')
            
            if '자치구' in df.columns:
                cnt = df['자치구'].value_counts().reset_index()
                cnt.columns = ['자치구명', '버스정류장_수']
                df_main = df_main.drop(columns=['버스정류장_수'], errors='ignore')
                df_main = df_main.merge(cnt, on='자치구명', how='left')
                status_log.append(f"✅ 버스 파일 적용됨 ({path_bus})")
        except: status_log.append("❌ 버스 파일 읽기 오류")
    else:
        status_log.append("⚠️ 버스 파일 없음 (샘플 데이터 사용)")

    # --- 지하철 병합 ---
    if path_sub:
        try:
            try: df = pd.read_csv(path_sub, encoding='utf-8')
            except: df = pd.read_csv(path_sub, encoding='cp949')
            
            if '자치구' in df.columns and '역개수' in df.columns:
                df = df.rename(columns={'자치구':'자치구명', '역개수':'지하철역_수'})
                df_main = df_main.drop(columns=['지하철역_수'], errors='ignore')
                df_main = df_main.merge(df[['자치구명', '지하철역_수']], on='자치구명', how='left')
                status_log.append(f"✅ 지하철 파일 적용됨 ({path_sub})")
        except: status_log.append("❌ 지하철 파일 읽기 오류")
    else:
        status_log.append("⚠️ 지하철 파일 없음 (샘플 데이터 사용)")

    # [3] 최종 병합 (지도 + 데이터)
    # 결측치 0 처리
    df_main = df_main.fillna(0)
    
    # 지도 데이터에 데이터프레임 병합
    if isinstance(gdf, geopandas.GeoDataFrame):
        final_gdf = gdf.merge(df_main, on='자치구명', how='left')
    else:
        final_gdf = df_main # 지도가 실패했으면 그냥 데이터프레임만 씀

    # [4] 지표 계산
    final_gdf['총_교통수단_수'] = final_gdf['버스정류장_수'] + final_gdf['지하철역_수']
    
    # 0 나누기 방지
    if '면적(km²)' in final_gdf.columns:
        final_gdf['인구 밀도 (명/km²)'] = final_gdf['총_상주인구_수'] / final_gdf['면적(km²)'].replace(0, 1)
    else:
        final_gdf['인구 밀도 (명/km²)'] = 0

    pop_safe = final_gdf['총_상주인구_수'].replace(0, 1)
    
    # **천명당 지표**
    final_gdf['대중교통 수 (인구 천명당)'] = (final_gdf['총_교통수단_수'] / pop_safe) * 1000
    final_gdf['교통 부족 순위 (위)'] = final_gdf['대중교통 수 (인구 천명당)'].rank(ascending=True, method='min')

    return final_gdf, status_log

# --------------------------------------------------------------------------
# 3. 데이터 로드 실행
# --------------------------------------------------------------------------
try:
    gdf, logs = load_data_safely()
    
    # 로그 출력
    with st.sidebar.expander("🛠️ 시스템 상태 로그"):
        for log in logs:
            st.write(log)
            
except Exception as e:
    st.error(f"치명적인 오류 발생: {e}")
    st.stop()

# 4. 메뉴 및 화면 표시

page_mode = st.sidebar.radio(
    "페이지 이동", 
    ["🏙️ 서울시 전체 분석", "🧩 유형별 4분면 분석", "📍 자치구별 상세 리포트"]

# --- 페이지 1: 전체 분석 ---
if page_mode == "🏙️ 서울시 전체 분석":
    st.header("🏙️ 서울시 전체 분석")
    
    col_name = '교통 부족 순위 (위)'
    if col_name in gdf.columns:
        st.subheader("🗺️ 교통 부족 순위 지도")
        
        # 지도 그리기 (GeoDataFrame인지 확인)
        if isinstance(gdf, geopandas.GeoDataFrame):
            fig = px.choropleth_mapbox(
                gdf, geojson=gdf.geometry.__geo_interface__, locations=gdf.index,
                color=col_name, mapbox_style="carto-positron", zoom=9.5,
                center={"lat": 37.5665, "lon": 126.9780}, opacity=0.6,
                hover_name='자치구명', color_continuous_scale='Reds_r'
            )
            fig.update_layout(height=500, margin={"r":0,"t":0,"l":0,"b":0})
            st.plotly_chart(fig, use_container_width=True)
        else:
            st.error("지도를 그릴 수 없습니다 (GeoJSON 로드 실패)")

        st.markdown("---")
        st.subheader("📊 자치구별 순위 그래프")
        
        df_sorted = gdf.sort_values(by=col_name, ascending=True) # 1위가 부족한 것
        fig_bar = px.bar(
            df_sorted, x='자치구명', y=col_name, 
            color=col_name, color_continuous_scale='Reds_r', text=col_name
        )
        st.plotly_chart(fig_bar, use_container_width=True)

# --- 페이지 2: 산점도 분석 ---
elif page_mode == "🧩 유형별 4분면 분석":
    st.header("🧩 유형별 4분면 분석")
    st.caption("X축: 수요(인구밀도) | Y축: 공급(천명당 교통수)")
    
    # 데이터 확인
    if '인구 밀도 (명/km²)' in gdf.columns:
        scat_fig = px.scatter(
            gdf, 
            x='인구 밀도 (명/km²)', 
            y='대중교통 수 (인구 천명당)', 
            text='자치구명',
            size='총_상주인구_수', 
            color='교통 부족 순위 (위)',
            color_continuous_scale='Reds_r',
            hover_data=['자치구명', '총_상주인구_수', '총_교통수단_수']
        )
        
        # 평균선
        mean_x = gdf['인구 밀도 (명/km²)'].mean()
        mean_y = gdf['대중교통 수 (인구 천명당)'].mean()
        
        scat_fig.add_vline(x=mean_x, line_dash="dash", annotation_text="평균 밀도")
        scat_fig.add_hline(y=mean_y, line_dash="dash", annotation_text="평균 공급")
        scat_fig.update_traces(textposition='top center')
        scat_fig.update_layout(height=600)
        
        st.plotly_chart(scat_fig, use_container_width=True)
        
        st.info("💡 **해석**: 오른쪽 아래(↘️)에 위치한 구들이 '인구는 많은데 교통은 부족한' 곳입니다.")
    else:
        st.error("분석에 필요한 데이터 컬럼이 없습니다.")

# --- 페이지 3: 상세 리포트 ---
else:
    st.header("📍 자치구별 상세 리포트")
    
    gu_list = gdf['자치구명'].unique()
    selected_gu = st.selectbox("자치구 선택", gu_list)
    
    row = gdf[gdf['자치구명'] == selected_gu].iloc[0]
    
    c1, c2, c3 = st.columns(3)
    c1.metric("인구", f"{int(row['총_상주인구_수']):,}명")
    c2.metric("천명당 교통수", f"{row['대중교통 수 (인구 천명당)']:.2f}개")
    c3.metric("부족 순위", f"{int(row['교통 부족 순위 (위)'])}위")
    
    st.success(f"**{selected_gu}** 데이터 분석 완료")
