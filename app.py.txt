import streamlit as st
import pandas as pd
import geopandas
import plotly.express as px
import os

# --------------------------------------------------------------------------
# 1. 페이지 설정
# --------------------------------------------------------------------------
st.set_page_config(layout="wide", page_title="서울시 도시계획 대시보드")

# --------------------------------------------------------------------------
# 2. 데이터 처리 헬퍼 함수 (숫자 깨짐 방지)
# --------------------------------------------------------------------------
def clean_numeric(value):
    """쉼표가 있는 문자열을 숫자로 변환합니다."""
    if isinstance(value, str):
        # 쉼표 제거 및 공백 제거
        value = value.replace(',', '').strip()
        try:
            return float(value)
        except:
            return 0
    return value

@st.cache_data(ttl=600)
def load_and_merge_data():
    data_info = {} 

    # [A] 지도 데이터 로드
    map_url = "https://raw.githubusercontent.com/southkorea/seoul-maps/master/kostat/2013/json/seoul_municipalities_geo_simple.json"
    try:
        gdf = geopandas.read_file(map_url)
        gdf = gdf.to_crs(epsg=4326)
        # 컬럼명 통일 (name -> 자치구명)
        if 'name' in gdf.columns: gdf = gdf.rename(columns={'name': '자치구명'})
        elif 'SIG_KOR_NM' in gdf.columns: gdf = gdf.rename(columns={'SIG_KOR_NM': '자치구명'})
        
        # 면적 계산 (도형 정보로 자동 계산)
        gdf_area = gdf.to_crs(epsg=5179)
        gdf['면적(km²)'] = gdf_area.geometry.area / 1_000_000
        data_info['지도'] = '✅ 로드 완료'
    except Exception as e:
        return None, {'error': str(e)}

    # [B] 초기 컬럼 생성 (에러 방지용 0값 채움)
    initial_cols = ['총_상주인구_수', '버스정류장_수', '지하철역_수', '총_교통수단_수']
    for c in initial_cols:
        gdf[c] = 0.0

    # 1. 인구 데이터 로드
    pop_file = '서울시 상권분석서비스(상주인구-자치구).csv'
    # 파일 경로 찾기
    pop_path = pop_file if os.path.exists(pop_file) else (f'./data/{pop_file}' if os.path.exists(f'./data/{pop_file}') else None)
    
    if pop_path:
        try:
            df_pop = pd.read_csv(pop_path, encoding='cp949')
            # 만약 숫자가 문자로 되어있다면 변환
            if '총_상주인구_수' in df_pop.columns:
                df_pop['총_상주인구_수'] = df_pop['총_상주인구_수'].apply(clean_numeric)
                
            grp = df_pop.groupby('자치구_코드_명')['총_상주인구_수'].mean().reset_index().rename(columns={'자치구_코드_명':'자치구명'})
            
            # 병합
            gdf = gdf.drop(columns=['총_상주인구_수'], errors='ignore')
            gdf = gdf.merge(grp, on='자치구명', how='left')
            data_info['인구'] = '✅ 로드 완료'
        except:
            data_info['인구'] = '❌ 읽기 오류'
    else:
        data_info['인구'] = '❌ 파일 없음'

    # 2. 버스 데이터 로드
    bus_file = "20250930기준_서울시정류소(정류소별).xlsx - 시내버스 정류장.csv"
    bus_path = bus_file if os.path.exists(bus_file) else (f'./data/{bus_file}' if os.path.exists(f'./data/{bus_file}') else None)
    
    if bus_path:
        try:
            try: df_bus = pd.read_csv(bus_path, encoding='utf-8')
            except: df_bus = pd.read_csv(bus_path, encoding='cp949')
            
            if '자치구' in df_bus.columns:
                # 자치구별 개수 세기
                bus_cnt = df_bus['자치구'].value_counts().reset_index()
                bus_cnt.columns = ['자치구명', '버스정류장_수']
                
                gdf = gdf.drop(columns=['버스정류장_수'], errors='ignore')
                gdf = gdf.merge(bus_cnt, on='자치구명', how='left')
                data_info['버스'] = '✅ 로드 완료'
            else:
                data_info['버스'] = '⚠️ 컬럼 없음'
        except:
            data_info['버스'] = '❌ 읽기 오류'
    else:
        # 파일이 없으면 기존 좌표 파일 검색
        bus_files = [f for f in os.listdir('./data') if 'Station' in f] if os.path.exists('./data') else []
        if bus_files: data_info['버스'] = '⚠️ 대체 파일 사용'
        else: data_info['버스'] = '❌ 파일 없음'

    # 3. 지하철 데이터 로드
    sub_file = "서울교통공사_자치구별 지하철역 정보_20250804.CSV"
    sub_path = sub_file if os.path.exists(sub_file) else (f'./data/{sub_file}' if os.path.exists(f'./data/{sub_file}') else None)
    
    if sub_path:
        try:
            try: df_sub = pd.read_csv(sub_path, encoding='utf-8')
            except: df_sub = pd.read_csv(sub_path, encoding='cp949')
            
            if '자치구' in df_sub.columns and '역개수' in df_sub.columns:
                df_sub = df_sub.rename(columns={'자치구': '자치구명', '역개수': '지하철역_수'})
                
                # 역개수도 숫자 변환 안전장치
                df_sub['지하철역_수'] = df_sub['지하철역_수'].apply(clean_numeric)
                
                gdf = gdf.drop(columns=['지하철역_수'], errors='ignore')
                gdf = gdf.merge(df_sub[['자치구명', '지하철역_수']], on='자치구명', how='left')
                data_info['지하철'] = '✅ 로드 완료'
            else:
                data_info['지하철'] = '⚠️ 컬럼 없음'
        except:
            data_info['지하철'] = '❌ 읽기 오류'
    else:
        data_info['지하철'] = '❌ 파일 없음'

    # 4. 결측치(NaN)를 0으로 채우기 (그래프 오류 방지 핵심)
    gdf = gdf.fillna(0)

    # 5. 최종 지표 계산
    gdf['총_교통수단_수'] = gdf['버스정류장_수'] + gdf['지하철역_수']
    gdf['인구 밀도 (명/km²)'] = gdf['총_상주인구_수'] / gdf['면적(km²)'].replace(0, 1)
    
    # [인구 천명당]
    pop_safe = gdf['총_상주인구_수'].replace(0, 1) # 0으로 나누기 방지
    gdf['대중교통 수 (인구 천명당)'] = (gdf['총_교통수단_수'] / pop_safe) * 1000
    
    # 순위 계산
    gdf['교통 부족 순위 (위)'] = gdf['대중교통 수 (인구 천명당)'].rank(ascending=True, method='min')

    return gdf, data_info

# --------------------------------------------------------------------------
# 3. 화면 구성
# --------------------------------------------------------------------------
result = load_and_merge_data()
if result is None or result[0] is None:
    st.error("데이터 로드 중 치명적인 오류가 발생했습니다.")
    st.stop()

gdf, data_info = result

# 메뉴 구성
page_mode = st.sidebar.radio(
    "페이지 이동", 
    ["🏙️ 서울시 전체 분석", "🧩 유형별 4분면 분석", "📍 자치구별 상세 리포트"]
)

# [디버깅용] 데이터 상태 확인창
with st.sidebar.expander("🛠️ 데이터 로드 상태 (디버깅)"):
    st.write(data_info)
    st.caption("그래프가 안 나오면 여기서 '인구'가 ✅인지 확인하세요.")
    if st.checkbox("원본 데이터 보기"):
        st.dataframe(gdf[['자치구명', '총_상주인구_수', '버스정류장_수', '지하철역_수']])

st.sidebar.markdown("---")

# --------------------------------------------------------------------------
# PAGE 1: 전체 분석 (지도 + 순위)
# --------------------------------------------------------------------------
if page_mode == "🏙️ 서울시 전체 분석":
    st.title("🏙️ 서울시 도시계획 종합 분석")
    
    # 분석 옵션
    options = [
        '총_상주인구_수', '인구 밀도 (명/km²)', 
        '버스정류장_수', '지하철역_수', 
        '대중교통 수 (인구 천명당)', '교통 부족 순위 (위)'
    ]
    selected_col = st.sidebar.selectbox("분석할 지표 선택", options, index=4)
    
    # 색상 설정
    if '부족' in selected_col: color_scale = 'Reds_r'
    elif '인구' in selected_col: color_scale = 'Blues'
    else: color_scale = 'Greens'

    st.subheader(f"🗺️ {selected_col} 지도 시각화")
    
    # 지도 그리기
    fig = px.choropleth_mapbox(
        gdf, geojson=gdf.geometry.__geo_interface__, locations=gdf.index,
        color=selected_col, mapbox_style="carto-positron", zoom=9.5,
        center={"lat": 37.5665, "lon": 126.9780}, opacity=0.6,
        hover_name='자치구명',
        color_continuous_scale=color_scale
    )
    fig.update_layout(margin={"r":0,"t":0,"l":0,"b":0}, height=500)
    st.plotly_chart(fig, use_container_width=True)

    st.markdown("---")
    st.subheader(f"📊 {selected_col} 자치구별 순위")
    
    # 순위 그래프
    df_sorted = gdf.sort_values(by=selected_col, ascending=('부족' not in selected_col))
    fig_bar = px.bar(
        df_sorted, x='자치구명', y=selected_col, 
        color=selected_col, color_continuous_scale=color_scale,
        text=selected_col
    )
    fig_bar.update_traces(texttemplate='%{text:,.2f}', textposition='outside')
    st.plotly_chart(fig_bar, use_container_width=True)

# --------------------------------------------------------------------------
# PAGE 2: 4분면 분석 (산점도)
# --------------------------------------------------------------------------
elif page_mode == "🧩 유형별 4분면 분석":
    st.title("🧩 도시 유형별 4분면 분석")
    st.info("X축: 수요(인구밀도)  |  Y축: 공급(천명당 교통수)")
    
    # 데이터 유효성 검사
    if gdf['총_상주인구_수'].sum() == 0:
        st.error("⚠️ 인구 데이터가 0입니다. 파일이 제대로 로드되지 않았습니다.")
        st.stop()

    # 산점도 그리기
    scat_fig = px.scatter(
        gdf, 
        x='인구 밀도 (명/km²)', 
        y='대중교통 수 (인구 천명당)', 
        text='자치구명',
        size='총_상주인구_수',  # 인구가 많으면 점이 커짐
        color='교통 부족 순위 (위)', 
        color_continuous_scale='Reds_r',
        hover_data=['총_상주인구_수', '총_교통수단_수'],
        size_max=40
    )

    # 평균선 추가 (4분면 기준)
    mean_x = gdf['인구 밀도 (명/km²)'].mean()
    mean_y = gdf['대중교통 수 (인구 천명당)'].mean()

    scat_fig.add_vline(x=mean_x, line_dash="dash", line_color="gray", annotation_text="평균 밀도")
    scat_fig.add_hline(y=mean_y, line_dash="dash", line_color="gray", annotation_text="평균 공급")

    scat_fig.update_traces(textposition='top center')
    scat_fig.update_layout(height=600, showlegend=False)
    
    st.plotly_chart(scat_fig, use_container_width=True)
    
    st.markdown("---")
    st.caption("💡 **해석 팁**: 오른쪽 아래(↘️)에 있는 구들이 **'사람은 많은데 교통은 부족한'** 집중 관리 지역입니다.")

# --------------------------------------------------------------------------
# PAGE 3: 상세 리포트
# --------------------------------------------------------------------------
else:
    st.title("📍 자치구별 상세 리포트")
    
    selected_gu = st.sidebar.selectbox("자치구 선택", gdf['자치구명'].unique())
    row = gdf[gdf['자치구명'] == selected_gu].iloc[0]
    
    # 지표 표시
    c1, c2, c3, c4 = st.columns(4)
    with c1: st.metric("인구", f"{int(row['총_상주인구_수']):,}명")

    fig = px.choropleth_mapbox(

    )
    fig.update_layout(showlegend=False, margin={"r":0,"t":0,"l":0,"b":0}, height=400)
    st.plotly_chart(fig, use_container_width=True)
